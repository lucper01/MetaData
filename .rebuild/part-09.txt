 exif['0th']=exif['0th']||{}; exif['Exif']=exif['Exif']||{}; exif['GPS']=exif['GPS']||{};
 const setOrDelete=(obj,key,val)=>{ if(val) obj[key]=val; else delete obj[key]; };

function crc32(bytes){ let c=0xffffffff; for(const b of bytes){ c ^= b; for(let k=0;k<8;k++) c=(c>>>1)^((c&1)?0xedb88320:0); } return (c^0xffffffff)>>>0; }
function u32(n){ return new Uint8Array([(n>>>24)&255,(n>>>16)&255,(n>>>8)&255,n&255]); }
function makePngChunk(type,data){ const tbytes=encodeText(type); const crcInput=new Uint8Array(tbytes.length+data.length); crcInput.set(tbytes); crcInput.set(data, tbytes.length); const out=new Uint8Array(12+data.length); out.set(u32(data.length),0); out.set(tbytes,4); out.set(data,8); out.set(u32(crc32(crcInput)), 8+data.length); return out; }
function makeTextChunk(key,val){ const kb=encodeText(key), vb=encodeText(val); const data=new Uint8Array(kb.length+1+vb.length); data.set(kb,0); data[kb.length]=0; data.set(vb,kb.length+1); return makePngChunk('tEXt', data); }
async function writePNG(meta, options={}){ const bytes=new Uint8Array(state.buffer); const chunks=parsePngChunks(state.buffer); const managed=new Set(['Title','Description','Author','Artist','Copyright','Software','Creation Time','DateTime','Keywords']); const parts=[bytes.slice(0,8)]; const values={ Title:meta.title||'', Description:meta.description||'', Author:options.cleanAuthor?'':(meta.author||''), Copyright:options.cleanAuthor?'':(meta.copyright||''), Software:options.cleanSoftware?'':(meta.modifierName||''), 'Creation Time':options.cleanDates?'':(meta.dateTime||''), Keywords:meta.keywords||'' }; let inserted=false; for(const c of chunks){ if(c.type==='tEXt'){ const zero=c.data.indexOf(0); const key=zero>=0 ? decodeText(c.data.slice(0,zero)) : ''; if(managed.has(key)) continue; } if(c.type==='IEND' && !inserted){ for(const [k,v] of Object.entries(values)){ if(v) parts.push(makeTextChunk(k,v)); } inserted=true; } parts.push(bytes.slice(c.start, c.start+12+c.length)); } return new Blob(parts, {type:'image/png'}); }
function makeId3TextFrame(id,text){ const payload=new Uint8Array(1+encodeText(text).length); payload[0]=3; payload.set(encodeText(text),1); const head=new Uint8Array(10); head.set(encodeText(id),0); head.set(u32be(payload.length),4); const out=new Uint8Array(head.length+payload.length); out.set(head); out.set(payload,head.length); return out; }
function makeCommentFrame(text){ const txt=encodeText(text); const payload=new Uint8Array(1+3+1+txt.length); payload[0]=3; payload.set(encodeText('eng'),1); payload[4]=0; payload.set(txt,5); const head=new Uint8Array(10); head.set(encodeText('COMM'),0); head.set(u32be(payload.length),4); const out=new Uint8Array(head.length+payload.length); out.set(head); out.set(payload,10); return out; }
async function writeMP3(meta, options={}){ const original=new Uint8Array(state.buffer); let audioStart=0; if(readAscii(original,0,3)==='ID3' && original.length>10){ const oldSize=synchsafeToInt(original[6],original[7],original[8],original[9]); audioStart=Math.min(original.length, 10+oldSize); } const frames=[]; const add=(id,key)=>{ let val=meta[key]||''; if(options.cleanAuthor && ['artist','albumArtist','copyright'].includes(key)) val=''; if(options.cleanDates && key==='year') val=''; if(val) frames.push(makeId3TextFrame(id,val)); }; add('TIT2','title'); add('TPE1','artist'); add('TALB','album'); add('TPE2','albumArtist'); add('TDRC','year'); add('TCON','genre'); add('TRCK','track'); add('TCOP','copyright'); add('TENC','modifierName'); if(meta.comment && !options.cleanAuthor) frames.push(makeCommentFrame(meta.comment)); const total=frames.reduce((a,b)=>a+b.length,0); const header=new Uint8Array(10); header.set(encodeText('ID3'),0); header[3]=3; header[4]=0; header[5]=0; header.set(intToSynchsafe(total),6); const out=new Uint8Array(10+total+(original.length-audioStart)); let p=0; out.set(header,p); p+=10; for(const f of frames){ out.set(f,p); p += f.length; } out.set(original.slice(audioStart),p); return new Blob([out], {type:'audio/mpeg'}); }
function setXmlText(doc,tag,value){ let el=doc.getElementsByTagName(tag)[0]; if(!el && value){ el=doc.createElement(tag); doc.documentElement.appendChild(el); } if(el) el.textContent = value || ''; }
