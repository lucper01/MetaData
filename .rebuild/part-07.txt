function detectType(file){ const ext=extOf(file.name), mime=mimeOf(file); if(ext==='jpg'||ext==='jpeg') return 'jpeg'; if(ext==='png') return 'png'; if(ext==='pdf') return 'pdf'; if(ext==='mp3') return 'mp3'; if(['docx','xlsx','xlsm','pptx'].includes(ext)) return 'office'; if(mime.startsWith('image/')) return 'generic-image'; if(mime.startsWith('audio/')) return 'generic-audio'; if(mime.startsWith('video/')) return 'generic-video'; return 'generic'; }
const descriptiveFieldSets = {
  jpeg:['title','description','artist','copyright','dateTime'],
  png:['title','description','author','copyright','dateTime','keywords'],
  pdf:['title','author','subject','keywords','creator'],
  mp3:['title','artist','album','albumArtist','year','genre','track','comment','copyright'],
  office:['title','subject','author','keywords','description','category'],
  generic:['title','description','author','keywords','comment','copyright'],
  'generic-image':['title','description','author','keywords','comment','copyright'],
  'generic-audio':['title','description','author','keywords','comment','copyright'],
  'generic-video':['title','description','author','keywords','comment','copyright']
};
const modifierFieldSets = {
  jpeg:['modifierName'],
  png:['modifierName'],
  pdf:['modifierName'],
  mp3:['modifierName'],
  office:['lastModifiedBy','modifierName'],
  generic:['modifierName'],
  'generic-image':['modifierName'],
  'generic-audio':['modifierName'],
  'generic-video':['modifierName']
};
const fieldSets = Object.fromEntries(Object.keys(descriptiveFieldSets).map(type => [type, [...descriptiveFieldSets[type], ...(modifierFieldSets[type]||[])]]));
function defaultMetaFor(type){ const out={}; (fieldSets[type]||fieldSets.generic).forEach(k=> out[k]=''); return out; }
function binFromBuf(buf){ const bytes=new Uint8Array(buf); let out=''; const step=0x8000; for(let i=0;i<bytes.length;i+=step) out += String.fromCharCode(...bytes.subarray(i, Math.min(i+step, bytes.length))); return out; }
function uint8FromBin(str){ const a=new Uint8Array(str.length); for(let i=0;i<str.length;i++) a[i]=str.charCodeAt(i)&255; return a; }
function decodeText(bytes){ try{ return new TextDecoder('utf-8').decode(bytes).replace(/\0/g,'').trim(); }catch{ return ''; } }
function encodeText(str){ return new TextEncoder().encode(str); }
function readAscii(bytes,start,len){ return String.fromCharCode(...bytes.slice(start,start+len)); }
function readU32BE(bytes,o){ return ((bytes[o]<<24)>>>0)+(bytes[o+1]<<16)+(bytes[o+2]<<8)+bytes[o+3]; }
function synchsafeToInt(b0,b1,b2,b3){ return (b0<<21)|(b1<<14)|(b2<<7)|b3; }
function intToSynchsafe(n){ return new Uint8Array([(n>>21)&0x7f,(n>>14)&0x7f,(n>>7)&0x7f,n&0x7f]); }
function u32be(n){ return new Uint8Array([(n>>>24)&255,(n>>>16)&255,(n>>>8)&255,n&255]); }
function detectImageDimensions(buffer){ const bytes=new Uint8Array(buffer); if(bytes.length>24 && readAscii(bytes,1,3)==='PNG') return `${readU32BE(bytes,16)} × ${readU32BE(bytes,20)}`; if(bytes[0]===0xFF && bytes[1]===0xD8){ let i=2; while(i<bytes.length){ if(bytes[i]!==0xFF){ i++; continue; } const marker=bytes[i+1]; const len=(bytes[i+2]<<8)+bytes[i+3]; if([0xC0,0xC1,0xC2,0xC3,0xC5,0xC6,0xC7,0xC9,0xCA,0xCB,0xCD,0xCE,0xCF].includes(marker)){ const h=(bytes[i+5]<<8)+bytes[i+6]; const w=(bytes[i+7]<<8)+bytes[i+8]; return `${w} × ${h}`; } i += 2 + len; } } return ''; }
async function loadFile(file){ try{ state.file=file; state.buffer=await file.arrayBuffer(); state.type=detectType(file); state.raw={}; state.meta=defaultMetaFor(state.type); state.original={}; if(state.previewUrl) URL.revokeObjectURL(state.previewUrl); state.previewUrl=URL.createObjectURL(file); await parseFile(); state.original=JSON.parse(JSON.stringify(state.meta)); state.dirty=false; addHistory(t('fileLoaded'), file.name); renderAll(); showPage('editor'); toast(t('fileLoaded'), t('fileLoadedText')); }catch(err){ console.error(err); toast(t('error'), err.message || String(err)); } }
async function parseFile(){
  const type=state.type;
  if(type==='jpeg'){ await parseJPEG(); }
  else if(type==='png'){ await parsePNG(); }
  else if(type==='pdf'){ await parsePDF(); }
  else if(type==='mp3'){ await parseMP3(); }
  else if(type==='office'){ await parseOffice(); }
  else { await parseGeneric(); }
}
async function parseGeneric(){ state.engine=t('engineGeneric'); state.mode=t('modeSidecar'); state.meta=defaultMetaFor(state.type); state.raw={ [t('mime')]: mimeOf(state.file), [t('dimensions')]: detectImageDimensions(state.buffer) }; }
async function parsePDF(){ const doc=await PDFLib.PDFDocument.load(state.buffer,{updateMetadata:false}); state.engine=t('enginePDF'); state.mode=t('modeEmbedded'); state.meta={ title:doc.getTitle()||'', author:doc.getAuthor()||'', subject:doc.getSubject()||'', keywords:Array.isArray(doc.getKeywords()) ? doc.getKeywords().join(', ') : (doc.getKeywords()||''), creator:doc.getCreator()||'', modifierName:doc.getProducer()||'' }; state.raw={ [t('title')]:state.meta.title, [t('author')]:state.meta.author, [t('subject')]:state.meta.subject, [t('keywords')]:state.meta.keywords, [t('creator')]:state.meta.creator, [t('modifierName')]:state.meta.modifierName, [t('pages')]: doc.getPageCount() }; }
async function parseJPEG(){ const binary=binFromBuf(state.buffer); let exif={}; try{ exif=piexif.load(binary); }catch{} const z=exif['0th']||{}, e=exif['Exif']||{}, gps=exif['GPS']||{}; const rv=(obj,key)=>obj[key]||''; state.engine=t('engineJPEG'); state.mode=t('modeEmbedded'); state.meta={ title:'', description:String(rv(z,piexif.ImageIFD.ImageDescription)||''), artist:String(rv(z,piexif.ImageIFD.Artist)||''), copyright:String(rv(z,piexif.ImageIFD.Copyright)||''), modifierName:String(rv(z,piexif.ImageIFD.Software)||''), dateTime:String(rv(e,piexif.ExifIFD.DateTimeOriginal)||rv(z,piexif.ImageIFD.DateTime)||'') }; state.raw={ [t('description')]:state.meta.description, [t('artist')]:state.meta.artist, [t('copyright')]:state.meta.copyright, [t('modifierName')]:state.meta.modifierName, [t('dateTime')]:state.meta.dateTime, [t('dimensions')]: detectImageDimensions(state.buffer) }; const make=rv(z,piexif.ImageIFD.Make), model=rv(z,piexif.ImageIFD.Model); if(make) state.raw[t('make')] = String(make); if(model) state.raw[t('model')] = String(model); if(Object.keys(gps).length) state.raw[t('gps')] = 'EXIF GPS'; }
function parsePngChunks(buffer){ const bytes=new Uint8Array(buffer); if(bytes.length<8 || readAscii(bytes,1,3)!=='PNG') return []; let pos=8, chunks=[]; while(pos+12<=bytes.length){ const len=readU32BE(bytes,pos), type=readAscii(bytes,pos+4,4), data=bytes.slice(pos+8,pos+8+len); chunks.push({type,data,start:pos,length:len}); pos += 12 + len; if(type==='IEND') break; } return chunks; }
async function parsePNG(){ const chunks=parsePngChunks(state.buffer); const text={}; for(const c of chunks){ if(c.type==='tEXt'){ const zero=c.data.indexOf(0); if(zero>=0){ const k=decodeText(c.data.slice(0,zero)); const v=decodeText(c.data.slice(zero+1)); text[k]=v; } } } state.engine=t('enginePNG'); state.mode=t('modeEmbedded'); state.meta={ title:text.Title||'', description:text.Description||'', author:text.Author||text.Artist||'', copyright:text.Copyright||'', modifierName:text.Software||'', dateTime:text['Creation Time']||text.DateTime||'', keywords:text.Keywords||'' }; state.raw={ ...text, [t('dimensions')]: detectImageDimensions(state.buffer) }; }
function decodeId3Text(data){ if(!data.length) return ''; const enc=data[0], body=data.slice(1); try{ if(enc===0) return new TextDecoder('iso-8859-1').decode(body).replace(/\0/g,'').trim(); if(enc===3) return new TextDecoder('utf-8').decode(body).replace(/\0/g,'').trim(); return new TextDecoder('utf-16').decode(body).replace(/\0/g,'').trim(); }catch{ return decodeText(body); } }
async function parseMP3(){ const bytes=new Uint8Array(state.buffer); const meta=defaultMetaFor('mp3'); const raw={}; if(readAscii(bytes,0,3)==='ID3' && bytes.length>10){ const ver=bytes[3], tagSize=synchsafeToInt(bytes[6],bytes[7],bytes[8],bytes[9]); let pos=10, end=Math.min(bytes.length, 10+tagSize); const map={TIT2:'title',TPE1:'artist',TALB:'album',TPE2:'albumArtist',TDRC:'year',TYER:'year',TCON:'genre',TRCK:'track',TCOP:'copyright',TENC:'modifierName'}; while(pos+10<=end){ const id=readAscii(bytes,pos,4); if(!/^[A-Z0-9]{4}$/.test(id)) break; let size = ver===4 ? synchsafeToInt(bytes[pos+4],bytes[pos+5],bytes[pos+6],bytes[pos+7]) : readU32BE(bytes,pos+4); if(size<=0 || pos+10+size>end) break; const data=bytes.slice(pos+10,pos+10+size); if(map[id]){ const val=decodeId3Text(data); meta[map[id]] = val; raw[id]=val; } else if(id==='COMM'){ const val=decodeId3Text(data.slice(4)); meta.comment=val; raw[id]=val; } pos += 10 + size; } } state.engine=t('engineMP3'); state.mode=t('modeEmbedded'); state.meta=meta; state.raw=raw; }
async function parseOffice(){ const zip=await JSZip.loadAsync(state.buffer); state.engine=t('engineOffice'); state.mode=t('modeEmbedded'); const meta=defaultMetaFor('office'); if(zip.file('docProps/core.xml')){ const xml=await zip.file('docProps/core.xml').async('text'); const doc=new DOMParser().parseFromString(xml,'application/xml'); const read=name=> doc.getElementsByTagName(name)[0]?.textContent || ''; meta.title=read('dc:title'); meta.subject=read('dc:subject'); meta.author=read('dc:creator'); meta.keywords=read('cp:keywords'); meta.description=read('dc:description'); meta.category=read('cp:category'); meta.lastModifiedBy=read('cp:lastModifiedBy'); const created=read('dcterms:created'); const modified=read('dcterms:modified'); if(created) state.raw['Created']=created; if(modified) state.raw['Modified']=modified; }
 if(zip.file('docProps/app.xml')){ const xml=await zip.file('docProps/app.xml').async('text'); const doc=new DOMParser().parseFromString(xml,'application/xml'); const read=name=> doc.getElementsByTagName(name)[0]?.textContent || ''; const company=read('Company'), manager=read('Manager'), application=read('Application'); if(company) state.raw[t('company')]=company; if(manager) state.raw[t('manager')]=manager; if(application){ state.raw[t('modifierName')]=application; meta.modifierName=application; } }
 state.meta=meta;
}
