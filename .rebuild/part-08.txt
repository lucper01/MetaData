function fieldEl(key,val){ const long=['description','keywords','comment'].includes(key); const id='fld-'+key; const ph = key==='keywords' ? (state.lang==='fr' ? 'Séparez les mots-clés par une virgule...' : 'Separate keywords with commas...') : ''; return `<div class="field ${long?'full':''}"><label for="${id}">${esc(t(key))}</label>${long ? `<textarea id="${id}" data-key="${key}" placeholder="${esc(ph)}">${esc(val||'')}</textarea>` : `<input id="${id}" data-key="${key}" value="${esc(val||'')}" placeholder="${esc(ph)}">`}</div>`; }
function renderEditor(){
  const empty=$('#editorEmpty'), ws=$('#editorWorkspace');
  if(!state.file){ empty.hidden=false; ws.hidden=true; $('#engineBadge').textContent='-'; return; }
  empty.hidden=true; ws.hidden=false;
  $('#engineBadge').textContent = state.engine;
  $('#editorSubtitle').textContent = `${state.file.name} · ${state.type.toUpperCase()}`;
  const mainKeys=descriptiveFieldSets[state.type] || descriptiveFieldSets.generic;
  const modifierKeys=modifierFieldSets[state.type] || modifierFieldSets.generic;
  $('#fieldGrid').innerHTML = mainKeys.map(k=> fieldEl(k, state.meta[k]||'')).join('');
  $('#modifierGrid').innerHTML = modifierKeys.map(k=> fieldEl(k, state.meta[k]||'')).join('');
  $('#modifierSection').hidden = modifierKeys.length===0;
  $('#saveModeBadge').textContent = state.mode;
  const embedded = state.mode===t('modeEmbedded');
  $('#compatNotice').className = 'notice ' + (embedded ? 'ok' : 'warn');
  $('#compatNotice').textContent = embedded ? t('embeddedInfo') : t('sidecarInfo');
  $('#modeExplainerTitle').textContent = embedded ? t('modeEmbeddedTitle') : t('modeSidecarTitle');
  $('#modeExplainerText').textContent = embedded ? t('modeEmbeddedText') : t('modeSidecarText');
  $('#downloadBtn').textContent = embedded ? t('save') : (state.lang==='fr' ? 'Exporter les métadonnées' : 'Export metadata');
  $$('#editorWorkspace [data-key]').forEach(el=> el.addEventListener('input', markDirty));
  updateEditorStatus();
  renderPreview(); renderFileInfo();
}
function markDirty(){ state.dirty=true; updateEditorStatus(); renderStats(); }
function updateEditorStatus(){ const el=$('#editorStatus'); if(!el) return; el.textContent = state.dirty ? t('statusChanged') : t('statusOriginal'); el.classList.toggle('dirty', !!state.dirty); el.classList.toggle('clean', !state.dirty); }
function renderPreview(){ const p=$('#preview'); if(!state.file){ p.innerHTML=''; return; } const mime=mimeOf(state.file); if(mime.startsWith('image/')) p.innerHTML = `<img alt="" src="${state.previewUrl}">`; else if(mime.startsWith('audio/')) p.innerHTML = `<audio controls src="${state.previewUrl}"></audio>`; else if(mime.startsWith('video/')) p.innerHTML = `<video controls src="${state.previewUrl}"></video>`; else if(state.type==='pdf') p.innerHTML = `<iframe src="${state.previewUrl}" title="PDF preview"></iframe>`; else p.innerHTML = `<div class="preview-empty"><div class="drop-icon">⌘</div><strong>${esc(state.file.name)}</strong><div class="badge" style="margin-top:12px">${esc(extOf(state.file.name).toUpperCase() || 'FILE')}</div></div>`; }
function renderFileInfo(){ if(!state.file){ $('#fileInfoList').innerHTML=''; return; } const data={ [t('mime')]:mimeOf(state.file), [t('size')]:humanSize(state.file.size), [t('extension')]:extOf(state.file.name)||'-', [t('modified')]:dateFmt(state.file.lastModified), [t('engine')]:state.engine, [t('saveMode')]:state.mode }; $('#fileInfoList').innerHTML = Object.entries(data).map(([k,v])=>`<div class="kv"><dt>${esc(k)}</dt><dd>${esc(v)}</dd></div>`).join(''); }
function renderCurrent(){ const box=$('#currentFileCard'); if(!state.file){ box.innerHTML = `<div class="empty"><strong>${esc(t('nothingLoaded'))}</strong><span>${esc(t('nothingLoadedText'))}</span></div>`; return; } const modeClass = state.mode===t('modeEmbedded') ? 'ok' : 'warn'; box.innerHTML = `<div style="display:grid;grid-template-columns:auto minmax(0,1fr) auto;gap:14px;align-items:center"><div class="brand-mark" style="width:56px;height:56px;font-size:.95rem">${esc((extOf(state.file.name)||'FILE').slice(0,4).toUpperCase())}</div><div><strong style="display:block;color:var(--dark);font:700 1.1rem Georgia,serif">${esc(state.file.name)}</strong><span style="display:block;color:var(--muted);font-size:.76rem;margin-top:4px">${esc(humanSize(state.file.size))} · ${esc(mimeOf(state.file))} · ${esc(state.engine)}</span></div><div><span class="badge ${modeClass}">${esc(state.mode)}</span></div></div>`; }
function renderRaw(){ const empty=$('#rawEmpty'), table=$('#rawTable'), tbody=table.querySelector('tbody'); if(!state.file){ empty.hidden=false; table.hidden=true; return; } empty.hidden=true; table.hidden=false; const q=$('#rawSearch').value.trim().toLowerCase(); const all={...state.raw, ...getCurrentMeta()}; const rows=Object.entries(all).filter(([k,v])=> !q || `${k} ${v}`.toLowerCase().includes(q)).sort((a,b)=>String(a[0]).localeCompare(String(b[0]))); tbody.innerHTML = rows.map(([k,v])=> `<tr><td>${esc(k)}</td><td>${esc(String(v??''))}</td></tr>`).join(''); }
function sensitiveItems(){ if(!state.file) return []; const all={...state.raw, ...getCurrentMeta()}; const rules=[ {label:t('identity'), keys:['author','artist','creator','lastmodified','company','manager','copyright']}, {label:t('gps'), keys:['gps','latitude','longitude','location']}, {label:t('device'), keys:['device','camera','model','make','software','producer','modifiername']}, {label:t('dates'), keys:['date','time','created','modified']} ]; const items=[]; for(const rule of rules){ for(const [k,v] of Object.entries(all)){ if(!v) continue; const key=String(k).toLowerCase(); if(rule.keys.some(x=> key.includes(x))){ items.push({label:rule.label, key:k, value:v}); } } } const seen=new Set(); return items.filter(i=>{ const id=`${i.label}|${i.key}|${i.value}`; if(seen.has(id)) return false; seen.add(id); return true; }); }
function renderPrivacy(){ const box=$('#privacyResults'); if(!state.file){ box.innerHTML = `<div class="empty"><span>${esc(t('privacyEmpty'))}</span></div>`; return; } const items=sensitiveItems(); if(!items.length){ box.innerHTML = `<div class="notice">${esc(t('noSensitive'))}</div>`; return; } box.innerHTML = items.map(i=> `<div class="history-item"><strong>${esc(i.label)} · ${esc(i.key)}</strong><span>${esc(String(i.value))}</span></div>`).join(''); }
function renderBatch(){ const box=$('#batchList'); if(!state.batch.length){ box.innerHTML = `<div class="empty"><span>${esc(t('batchEmpty'))}</span></div>`; return; } box.innerHTML = state.batch.map((f,i)=> `<div class="batch-row"><span class="badge">${esc((extOf(f.name)||'?').toUpperCase())}</span><div><strong>${esc(f.name)}</strong><span>${esc(humanSize(f.size))} · ${esc(mimeOf(f))}</span></div><button class="button small" data-rm="${i}">×</button></div>`).join(''); $$('[data-rm]').forEach(btn=> btn.onclick=()=>{ state.batch.splice(Number(btn.dataset.rm),1); renderBatch(); }); }
function renderHistory(){ const box=$('#historyList'); if(!state.history.length){ box.innerHTML = `<div class="empty"><span>-</span></div>`; return; } box.innerHTML = state.history.map(h=> `<div class="history-item"><strong>${esc(h.action)}${h.file ? ' · ' + esc(h.file) : ''}</strong><span>${esc(dateFmt(h.time))}</span></div>`).join(''); }
function renderStats(){ const fields = Object.entries({...state.raw, ...getCurrentMeta()}).filter(([,v])=>v!==''&&v!==null&&v!==undefined).length; const editable = state.file ? (fieldSets[state.type] || fieldSets.generic).length : 0; $('#statFields').textContent = String(fields); $('#statEditable').textContent = String(editable); $('#statSensitive').textContent = String(sensitiveItems().length); $('#statMode').textContent = state.file ? (state.mode===t('modeEmbedded') ? (state.lang==='fr' ? 'Intégré' : 'Embedded') : 'Sidecar') : '-'; }
function renderHeader(){ if(state.file){ $('#topTitle').textContent = state.file.name; $('#topSubtitle').textContent = `${state.type.toUpperCase()} · ${state.engine} · ${state.mode}`; $('#saveBtn').disabled=false; $('#resetBtn').disabled=false; $('#exportJsonTop').disabled=false; } else { $('#topTitle').textContent='MetaData'; $('#topSubtitle').textContent=t('topSubtitleDefault'); $('#saveBtn').disabled=true; $('#resetBtn').disabled=true; $('#exportJsonTop').disabled=true; } }
function renderAll(){ renderHeader(); renderCurrent(); renderEditor(); renderRaw(); renderPrivacy(); renderBatch(); renderHistory(); renderStats(); }
async function saveCurrent(cleanOptions=null){ if(!state.file){ toast(t('error'), t('selectFileFirst')); return; } try{ if(state.mode===t('modeEmbedded')){ const blob=await buildEmbeddedBlob(cleanOptions); downloadBlob(blob, buildOutputName(state.file.name, cleanOptions ? '_clean' : '_metadata')); addHistory(cleanOptions ? t('cleanDone') : t('saveDone'), state.file.name); if(!cleanOptions){ state.dirty=false; updateEditorStatus(); } toast(t('done'), cleanOptions ? t('cleanDone') : t('saveDoneText')); } else { const sidecarBlob=buildSidecarJson(cleanOptions); downloadBlob(sidecarBlob, state.file.name + (cleanOptions ? '.clean.metadata.json' : '.metadata.json')); addHistory(cleanOptions ? t('cleanDone') : t('saveDone'), state.file.name); if(!cleanOptions){ state.dirty=false; updateEditorStatus(); } toast(t('done'), cleanOptions ? t('cleanDone') : t('saveDoneText')); } }catch(err){ console.error(err); toast(t('error'), err.message || String(err)); } }
function buildOutputName(name,suffix){ const i=name.lastIndexOf('.'); return i>=0 ? name.slice(0,i)+suffix+name.slice(i) : name+suffix; }
function downloadBlob(blob,name){ const url=URL.createObjectURL(blob); const a=document.createElement('a'); a.href=url; a.download=name; document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),1200); }
function cleanMeta(meta, options){ const out=JSON.parse(JSON.stringify(meta)); if(options){ if(options.cleanAuthor){ ['author','artist','albumArtist','creator','lastModifiedBy','copyright'].forEach(k=> { if(k in out) out[k]=''; }); } if(options.cleanSoftware){ ['software','producer','modifierName'].forEach(k=> { if(k in out) out[k]=''; }); } if(options.cleanDates){ ['dateTime','year'].forEach(k=> { if(k in out) out[k]=''; }); } } return out; }
